cmake_minimum_required(VERSION 3.18)
project(tianji_bridge LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/prepare_sdk.py
                ${CMAKE_BINARY_DIR} RESULT_VARIABLE prepared)
if(NOT prepared EQUAL 0)
  message(FATAL_ERROR "Could not prepare pinned Tianji SDK")
endif()
set(VENDOR ${CMAKE_BINARY_DIR}/vendor)
file(GLOB CONTROL_SOURCES CONFIGURE_DEPENDS ${VENDOR}/contrlSDK100343/*.cpp)
file(GLOB KINE_SOURCES CONFIGURE_DEPENDS ${VENDOR}/kinematicsSDK/*.cpp)
add_library(tianji_kine SHARED ${KINE_SOURCES})
target_compile_definitions(tianji_kine PRIVATE CMPL_LIN)
target_link_options(tianji_kine PRIVATE -Wl,-Bsymbolic)
add_library(tianji_bridge SHARED bridge.cpp kinematics.cpp ${CONTROL_SOURCES})
target_compile_definitions(tianji_bridge PRIVATE CMPL_LIN)
target_include_directories(tianji_bridge PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ${VENDOR}/contrlSDK100343)
target_include_directories(tianji_bridge PRIVATE ${VENDOR})
target_link_libraries(tianji_bridge PRIVATE tianji_kine pthread rt)
target_link_options(tianji_bridge PRIVATE -Wl,-Bsymbolic)
set_target_properties(tianji_bridge PROPERTIES BUILD_RPATH "$ORIGIN" INSTALL_RPATH "$ORIGIN")
include(CTest)
if(BUILD_TESTING)
  add_executable(test_tianji_native test_bridge.cpp)
  target_include_directories(test_tianji_native PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ${VENDOR}/contrlSDK100343)
  target_compile_definitions(test_tianji_native PRIVATE CMPL_LIN)
  target_compile_options(test_tianji_native PRIVATE -fPIC)
  target_link_libraries(test_tianji_native PRIVATE tianji_bridge)
  add_test(NAME tianji_native COMMAND test_tianji_native)
  set_tests_properties(tianji_native PROPERTIES TIMEOUT 20
                      FAIL_REGULAR_EXPRESSION "system version:|Robot released")
endif()
